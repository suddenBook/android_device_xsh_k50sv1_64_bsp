#!/usr/bin/env python3
"""Exercise the actual old/new one-line WLAN dump with kernel printf seams."""
import hashlib
import json
import pathlib
import subprocess

ROOT = pathlib.Path(__file__).resolve().parent
REPO = ROOT.parent / "build/project/vendor/mediatek/kernel_modules/connectivity"
SOURCE = "wlan/core/gen2/common/dump.c"
BASELINE = "01aa9ae"
OUT = ROOT / "dump-boundary"
OUT.mkdir(exist_ok=True)


def function(text):
    start = text.index("VOID dumpMemory8IEOneLine(")
    brace = text.index("{", start)
    depth = 0
    for end in range(brace, len(text)):
        if text[end] == "{":
            depth += 1
        elif text[end] == "}":
            depth -= 1
            if depth == 0:
                return text[start : end + 1]
    raise ValueError("Unterminated source function")


SEAM = r"""
#include <assert.h>
#include <limits.h>
#include <stdarg.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define VOID void
#define IN
#define UINT_8 unsigned char
#define UINT_32 uint32_t
#define PUINT_8 unsigned char *
#define BUF_MSG_LENGTH 1024
#define CFG_IE_BUFFER_SIZE 512
#define ASSERT(x) assert(x)
static unsigned char *active_buf;
static size_t active_size;
static unsigned bad_size, bad_pointer;
static char logged[BUF_MSG_LENGTH + 2];

static void zero_buffer(void *ptr, size_t size)
{
    active_buf = ptr;
    active_size = size;
    memset(ptr, 0, size);
}
#define kalMemZero zero_buffer

static int append_printf(unsigned char *dst, size_t size, int actual,
                         const char *format, va_list ap)
{
    uintptr_t address = (uintptr_t)dst, base = (uintptr_t)active_buf;
    int invalid = address < base || address - base >= active_size;
    int count;
    bad_pointer += invalid;
    /* Same early rejection as kernel 3.18 vsnprintf, before a destination write. */
    if (size > INT_MAX) {
        bad_size++;
        return 0;
    }
    assert(!invalid);
    assert(size <= active_size - (address - base));
    if (strcmp(format, "[%pM],Len:%u:[") == 0) {
        unsigned char *mac = va_arg(ap, unsigned char *);
        unsigned length = va_arg(ap, unsigned);
        count = snprintf((char *)dst, size,
                         "[%02x:%02x:%02x:%02x:%02x:%02x],Len:%u:[",
                         mac[0], mac[1], mac[2], mac[3], mac[4], mac[5], length);
    } else {
        count = vsnprintf((char *)dst, size, format, ap);
    }
    assert(count >= 0);
    if (actual)
        return size ? (count < (int)size ? count : (int)size - 1) : 0;
    return count;
}

static int __attribute__((unused))
required_printf(unsigned char *dst, size_t size, const char *format, ...)
{
    int count;
    va_list ap;
    va_start(ap, format);
    count = append_printf(dst, size, 0, format, ap);
    va_end(ap);
    return count;
}

static int __attribute__((unused))
stored_printf(unsigned char *dst, size_t size, const char *format, ...)
{
    int count;
    va_list ap;
    va_start(ap, format);
    count = append_printf(dst, size, 1, format, ap);
    va_end(ap);
    return count;
}

#define kalSnprintf required_printf
#define scnprintf stored_printf

static void log_line(const char *format, const unsigned char *text)
{
    assert(strcmp(format, "%s\n") == 0);
    assert(memchr(text, 0, active_size) != NULL);
    snprintf(logged, sizeof(logged), "%s", text);
}
#define LOG_FUNC log_line
"""

CHECK = r"""
int main(void)
{
    const uint32_t lengths[] = {
        0, 1, 2, 9, 99, 329, 330, 331, 332, 333, 511, 512, 513, 952, UINT32_MAX
    };
    unsigned char mac[] = {0x52, 0x65, 0x67, 0x49, 0x6e, 0x66};
    unsigned char input[512];
    unsigned failures = 0, index;
    for (index = 0; index < sizeof(input); index++)
        input[index] = index;
    for (index = 0; index < sizeof(lengths) / sizeof(lengths[0]); index++) {
        uint32_t length = lengths[index];
        char prefix[64];
        const char *payload;
        size_t n, payload_len;
        int valid = 1;
        bad_size = bad_pointer = 0;
        dumpMemory8IEOneLine(mac, input, length);
        snprintf(prefix, sizeof(prefix), "[52:65:67:49:6e:66],Len:%u:[", length);
        valid &= strncmp(logged, prefix, strlen(prefix)) == 0;
        n = strlen(logged);
        valid &= n < BUF_MSG_LENGTH && n > 0 && logged[n - 1] == ']';
        payload = logged + strlen(prefix);
        payload_len = n >= strlen(prefix) + 1 ? n - strlen(prefix) - 1 : 0;
        valid &= payload_len % 3 == 0;
        if (valid) {
            size_t bytes = payload_len / 3, i;
            size_t capacity = (BUF_MSG_LENGTH - strlen(prefix) - 2) / 3;
            size_t expected = length < 512 ? length : 512;
            if (expected > capacity)
                expected = capacity;
            valid &= bytes == expected;
            for (i = 0; i < bytes; i++) {
                char triplet[4];
                snprintf(triplet, sizeof(triplet), "%02x,", input[i]);
                valid &= memcmp(payload + 3 * i, triplet, 3) == 0;
            }
        }
        valid &= bad_size == 0 && bad_pointer == 0;
        failures += !valid;
        printf("length=%u size_rejections=%u invalid_destinations=%u output=%zu valid=%d\n",
               length, bad_size, bad_pointer, n, valid);
    }
    printf("cases=%zu failures=%u\n", sizeof(lengths) / sizeof(lengths[0]), failures);
    return failures ? 1 : 0;
}
"""


def exercise(label, text):
    body = function(text)
    c_file = OUT / (label + ".c")
    binary = OUT / label
    c_file.write_text(SEAM + "\n" + body + "\n" + CHECK)
    command = [
        "cc", "-std=c11", "-Wall", "-Wextra", "-Werror", "-O1", "-g",
        "-fsanitize=address,undefined", str(c_file), "-o", str(binary)
    ]
    subprocess.run(command, check=True)
    run = subprocess.run([str(binary)], capture_output=True, text=True)
    (OUT / (label + ".txt")).write_text(run.stdout + run.stderr)
    return {
        "source_sha256": hashlib.sha256(text.encode()).hexdigest(),
        "function_sha256": hashlib.sha256(body.encode()).hexdigest(),
        "compile": command,
        "returncode": run.returncode,
        "stdout": run.stdout,
        "stderr": run.stderr,
    }


old = subprocess.check_output(
    ["git", "-C", str(REPO), "show", BASELINE + ":" + SOURCE], text=True
)
new = (REPO / SOURCE).read_text()
results = {
    "baseline_commit": subprocess.check_output(
        ["git", "-C", str(REPO), "rev-parse", BASELINE], text=True
    ).strip(),
    "old": exercise("old", old),
    "new": exercise("new", new),
}
(OUT / "results.json").write_text(json.dumps(results, indent=2) + "\n")
print("old:\n" + results["old"]["stdout"])
print("new:\n" + results["new"]["stdout"])
assert results["old"]["returncode"] == 1, results["old"]
assert "length=952 size_rejections=181 invalid_destinations=181" in results["old"]["stdout"]
assert results["new"]["returncode"] == 0, results["new"]
assert not results["new"]["stderr"], results["new"]
